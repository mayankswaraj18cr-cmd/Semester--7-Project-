▸ Python Source Code

graph = {
    'A': {'B': 2, 'C': 4},
    'B': {'A': 2, 'D': 3},
    'C': {'A': 4, 'D': 1},
    'D': {'B': 3, 'C': 1}
}

def show_routes():
    for node in graph:
        print(node, "->", graph[node])

show_routes()

# Output:
# A -> {'B': 2, 'C': 4}
# B -> {'A': 2, 'D': 3}
# C -> {'A': 4, 'D': 1}
# D -> {'B': 3, 'C': 1}
